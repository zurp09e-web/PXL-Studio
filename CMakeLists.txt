cmake_minimum_required(VERSION 3.15)
project(pxl_native LANGUAGES CXX)
add_library(pxl_native SHARED native/pxl_native.cpp native/pxl_v8.cpp)
target_compile_features(pxl_native PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(pxl_native PRIVATE /O2 /EHsc)
  if(CMAKE_SIZEOF_VOID_P EQUAL 8 AND CMAKE_SYSTEM_PROCESSOR MATCHES "AMD64|x86_64|X86")
    target_sources(pxl_native PRIVATE native/pxl_v8_msvc_avx2.cpp)
    set_source_files_properties(native/pxl_v8_msvc_avx2.cpp PROPERTIES COMPILE_OPTIONS "/arch:AVX2")
    target_compile_definitions(pxl_native PRIVATE PXL_HAS_MSVC_AVX2)
    # AVX-512 is optional at build time because not every MSVC toolchain has
    # matching code-generation support; the runtime dispatcher guards it.
    option(PXL_ENABLE_AVX512 "Build the optional MSVC AVX-512BW kernel" OFF)
    if(PXL_ENABLE_AVX512)
      target_sources(pxl_native PRIVATE native/pxl_v8_msvc_avx512.cpp)
      set_source_files_properties(native/pxl_v8_msvc_avx512.cpp PROPERTIES COMPILE_OPTIONS "/arch:AVX512")
      target_compile_definitions(pxl_native PRIVATE PXL_HAS_MSVC_AVX512)
    endif()
  endif()
else()
  target_compile_options(pxl_native PRIVATE -O3)
endif()
set_target_properties(pxl_native PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
